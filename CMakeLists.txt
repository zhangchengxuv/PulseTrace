cmake_minimum_required(VERSION 3.24)
project(PulseTrace VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(mouse_core src/mouse_monitor.cpp)
target_include_directories(mouse_core PUBLIC include)
if(MSVC)
  target_compile_options(mouse_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(mouse_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
option(PULSETRACE_BUILD_DESKTOP "Build Win32/WebView2 desktop application" ON)
if(WIN32 AND PULSETRACE_BUILD_DESKTOP)
  enable_language(RC)
  set(PULSETRACE_DEPS "${CMAKE_SOURCE_DIR}/build/deps" CACHE PATH "Pinned dependency directory")
  set(WEBVIEW2_SDK_ROOT "${PULSETRACE_DEPS}/webview2" CACHE PATH "Extracted WebView2 NuGet SDK")
  if(NOT EXISTS "${WEBVIEW2_SDK_ROOT}/build/native/include/WebView2.h" OR NOT EXISTS "${PULSETRACE_DEPS}/json.hpp")
    message(FATAL_ERROR "Desktop dependencies missing. Run tools/bootstrap-deps.ps1 or set PULSETRACE_BUILD_DESKTOP=OFF")
  endif()
  add_library(sqlite STATIC "${PULSETRACE_DEPS}/sqlite-amalgamation-3540100/sqlite3.c")
  target_include_directories(sqlite PUBLIC "${PULSETRACE_DEPS}/sqlite-amalgamation-3540100")
  target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_OMIT_LOAD_EXTENSION)
  add_library(product_core src/keyboard_monitor.cpp src/database.cpp src/network_monitor.cpp src/core_service.cpp src/protocol.cpp)
  target_include_directories(product_core PUBLIC include "${PULSETRACE_DEPS}")
  target_link_libraries(product_core PUBLIC raw_input sqlite iphlpapi ws2_32 ole32 uuid)
  add_executable(PulseTrace WIN32 src/desktop/main.cpp src/desktop/webview_host.cpp src/desktop/PulseTrace.rc)
  target_include_directories(PulseTrace PRIVATE "${CMAKE_SOURCE_DIR}/src/desktop")
  target_include_directories(PulseTrace PRIVATE "${WEBVIEW2_SDK_ROOT}/build/native/include")
  target_link_libraries(PulseTrace PRIVATE product_core ole32 shell32 user32 advapi32 gdi32 comdlg32 shlwapi uuid)
  if(MINGW)
    target_compile_options(PulseTrace PRIVATE -fms-extensions -Wno-language-extension-token)
    target_link_options(PulseTrace PRIVATE -municode)
  endif()
  add_custom_target(desktop_assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web/desktop" "$<TARGET_FILE_DIR:PulseTrace>/web/desktop"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WEBVIEW2_SDK_ROOT}/runtimes/win-x64/native/WebView2Loader.dll" "$<TARGET_FILE_DIR:PulseTrace>/WebView2Loader.dll")
  add_dependencies(PulseTrace desktop_assets)
endif()
if(MINGW)
  target_link_options(mouse_core INTERFACE -static)
endif()
if(WIN32)
  add_library(raw_input src/raw_input_host.cpp src/probe_report.cpp)
  target_include_directories(raw_input PUBLIC include)
  target_compile_definitions(raw_input PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(raw_input PUBLIC mouse_core user32 psapi wtsapi32)
  add_executable(mouse_probe src/mouse_probe.cpp)
  target_link_libraries(mouse_probe PRIVATE raw_input)
  add_executable(mouse_benchmark tests/mouse_benchmark.cpp)
  target_link_libraries(mouse_benchmark PRIVATE raw_input)
  add_executable(mouse_latency_target tests/mouse_latency_target.cpp)
  target_compile_definitions(mouse_latency_target PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(mouse_latency_target PRIVATE mouse_core user32 gdi32)
  add_executable(mouse_dashboard src/mouse_dashboard.cpp)
  target_compile_definitions(mouse_dashboard PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(mouse_dashboard PRIVATE raw_input ws2_32)
  add_custom_target(dashboard_assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:mouse_dashboard>/web")
  add_dependencies(mouse_dashboard dashboard_assets)
  if(TARGET desktop_assets)
    add_dependencies(desktop_assets dashboard_assets)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mouse_tests tests/mouse_tests.cpp)
  target_link_libraries(mouse_tests PRIVATE mouse_core)
  add_test(NAME mouse_statistics COMMAND mouse_tests)
  if(TARGET product_core)
    add_executable(product_tests tests/product_tests.cpp)
    target_link_libraries(product_tests PRIVATE product_core shell32)
    add_executable(database_crash_writer tests/database_crash_writer.cpp)
    target_link_libraries(database_crash_writer PRIVATE product_core)
    add_dependencies(product_tests database_crash_writer)
    add_executable(network_capability_probe tests/network_capability_probe.cpp)
    target_link_libraries(network_capability_probe PRIVATE product_core advapi32)
    add_test(NAME product_statistics_storage_protocol_lifecycle COMMAND product_tests)
    set_tests_properties(product_statistics_storage_protocol_lifecycle PROPERTIES TIMEOUT 40)
    add_test(NAME desktop_webview2_lifecycle COMMAND PulseTrace --self-test --data-dir "${CMAKE_BINARY_DIR}/data/tests/desktop-smoke")
    set_tests_properties(desktop_webview2_lifecycle PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  endif()
endif()
